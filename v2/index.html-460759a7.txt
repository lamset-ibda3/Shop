      { id: "s2", worker_id: "w2", worker: { full_name: "محمود" }, opened_at: new Date(now - 30 * H).toISOString(), ended_at: new Date(now - 22 * H).toISOString(), closed_at: null, opening_cash: 200, counted_cash: null },
      { id: "s3", worker_id: "w3", worker: { full_name: "سارة" }, opened_at: new Date(now - 54 * H).toISOString(), ended_at: new Date(now - 46 * H).toISOString(), closed_at: new Date(now - 45 * H).toISOString(), opening_cash: 250, counted_cash: 1840 }
    ]
  };
  const q = new URLSearchParams(location.search).get("as");
  if (q) db.me = db.team.find(x => q === "owner" ? x.id === "o" : x.id === "w1");
  const wait = v => new Promise(r => setTimeout(() => r(v), 60));
  api = {
    async session() { return db.me ? { id: db.me.id } : null; },
    async signIn(u) { db.me = db.team.find(x => x.full_name === u.trim()) || db.team[0]; },
    async signOut() { db.me = null; },
    async me() { return db.me; },
    async products() { return wait(db.products.map(({ _cost, ...r }) => r)); },
    async costs() { return wait(Object.fromEntries(db.products.map(p => [p.id, p._cost]))); },
    async saveProduct(p) {
      if (p.id) { Object.assign(db.products.find(x => x.id === p.id), { name: p.name, price: p.price, _cost: p.cost, stock: p.stock, active: p.active }); return p.id; }
      const id = "p" + Date.now(); db.products.push({ id, name: p.name, code: String(100001 + db.products.length), price: p.price, stock: p.stock, active: p.active, _cost: p.cost }); return id;
    },
    async addPurchase(x) {
      let tot = 0; const items = x.items.map(i => { const p = db.products.find(z => z.id === i.product_id); p.stock += i.qty; p._cost = i.unit_cost; tot += i.qty * i.unit_cost; return { qty: i.qty, unit_cost: i.unit_cost, products: { name: p.name } }; });
      db.purchases.unshift({ id: "b" + Date.now(), supplier: x.supplier, note: x.note, total: tot, created_at: new Date().toISOString(), purchase_items: items });
    },
    async purchases() { return wait(db.purchases); },
    async shifts() { return wait(db.me.role === "owner" ? db.shifts : db.shifts.filter(s => s.worker_id === db.me.id)); },
    async openShift(c) { db.shifts.unshift({ id: "s" + Date.now(), worker_id: db.me.id, worker: { full_name: db.me.full_name }, opened_at: new Date().toISOString(), ended_at: null, closed_at: null, opening_cash: c, counted_cash: null }); },
    async endShift() { const s = db.shifts.find(x => x.worker_id === db.me.id && !x.ended_at); if (s) s.ended_at = new Date().toISOString(); },
    async closeShift(id, c, n) { const s = db.shifts.find(x => x.id === id); s.ended_at = s.ended_at || new Date().toISOString(); s.closed_at = new Date().toISOString(); s.counted_cash = c; s.note = n; },
    async team() { return db.team; },
    async updateMember(id, patch) { Object.assign(db.team.find(x => x.id === id), patch); }
  };
}

function memberSheet(id) {
  const m = S.team.find(x => x.id === id), self = m.id === S.me.id;
  sheet("حساب " + m.full_name, `
    <div class="field"><label>الاسم</label><input class="input" id="mn" value="${esc(m.full_name)}"></div>
    ${self ? "" : `<div class="row between" style="margin-bottom:18px"><span>الحساب شغّال</span><button type="button" class="switch ${m.active ? "on" : ""}" id="ma"></button></div>`}
    <button class="btn block" id="msave">حفظ</button>`, el => {
    $("#ma", el)?.addEventListener("click", e => e.currentTarget.classList.toggle("on"));
    $("#msave", el).onclick = async e => {
      const name = $("#mn", el).value.trim(); if (!name) return toast("اكتب الاسم", true);
      e.currentTarget.disabled = true;
      try { await api.updateMember(id, self ? { full_name: name } : { full_name: name, active: $("#ma", el).classList.contains("on") }); closeSheet(); await loadAll(); toast("اتحفظ ✓"); render(); }
      catch (er) { toast(errMsg(er), true); e.currentTarget.disabled = false; }
    };
  });
}

function labelsSheet() {
  const prods = S.products.filter(p => p.active), sel = new Set(prods.map(p => p.id));
  sheet("ملصقات الباركود", `
    <p class="muted small" style="margin-top:-8px">اختار المنتجات واكتب كام ملصق لكل منتج، وبعدها اطبع على ورق ملصقات. كل ملصق عليه الاسم والباركود والسعر.</p>
    <div class="field"><label>عدد الملصقات لكل منتج</label><input class="input" id="lc" inputmode="numeric" value="${DEMO ? 2 : 10}"></div>
    <div class="list" style="margin-bottom:16px">${prods.map(p => `<label class="item" style="cursor:pointer"><input type="checkbox" data-id="${esc(p.id)}" checked style="width:20px;height:20px;accent-color:var(--primary)"><div class="grow"><div class="t">${esc(p.name)}</div><div class="s">كود ${esc(p.code)}</div></div></label>`).join("")}</div>
    <button class="btn block" id="lp">${ic("tag")} تجهيز الطباعة</button>`, el => {
    el.addEventListener("change", e => { const id = e.target.dataset.id; if (id) e.target.checked ? sel.add(id) : sel.delete(id); });
    $("#lp", el).onclick = () => {
      const n = Math.max(1, Math.round(num($("#lc", el).value)) || 1), pa = $("#print-area");
      pa.innerHTML = prods.filter(p => sel.has(p.id)).map(p => Array.from({ length: n }, () => `<div class="lab"><div class="n">${esc(p.name)}</div><svg data-code="${esc(p.code)}"></svg><div class="p">${money(p.price)} ج.م</div></div>`).join("")).join("");
      if (!pa.children.length) return toast("اختار منتج واحد على الأقل", true);
      pa.querySelectorAll("svg[data-code]").forEach(s => { try { JsBarcode(s, s.dataset.code, { format: "CODE128", displayValue: true, fontSize: 14, margin: 0, height: 50, width: 2 }); } catch (_) { } });
      setTimeout(() => window.print(), 150);
    };
  });
}

const VIEWS = { home: viewHome, products: viewProducts, purchases: viewPurchases, shifts: viewShifts, team: viewTeam };
function render() {
  const root = $("#root");
  if (!S.me) { root.innerHTML = viewLogin(); return; }
  let r = (location.hash.replace("#/", "") || "home");
  if (!VIEWS[r] || (["purchases", "team"].includes(r) && !isOwner())) r = "home";
  S.route = r;
  root.innerHTML = S.loaded ? VIEWS[r]() : `<div class="empty" style="padding-top:30vh"><b>لحظة...</b></div>`;
}
window.addEventListener("hashchange", () => { closeSheet(); render(); window.scrollTo(0, 0); });
window.addEventListener("online", render); window.addEventListener("offline", render);

document.addEventListener("click", async e => {
  const b = e.target.closest("[data-act]"); if (!b) return;
  const a = b.dataset.act, id = b.dataset.id;
  try {
    if (a === "theme") { setTheme(document.documentElement.dataset.theme === "dark" ? "light" : "dark"); render(); }
    else if (a === "logout") { await api.signOut(); Object.assign(S, { me: null, user: null, loaded: false }); render(); }
    else if (a === "openshift") { b.disabled = true; await api.openShift(num($("#oc")?.value)); await loadAll(); toast("اتفتحت الوردية ✓"); render(); }
    else if (a === "endshift") { if (!confirm("تنهي الوردية؟")) return; await api.endShift(); await loadAll(); toast("اتنهت الوردية. المالك هيراجعها."); render(); }
    else if (a === "newp") productSheet(null);
    else if (a === "editp" && isOwner()) productSheet(id);
    else if (a === "newbuy") purchaseSheet();
    else if (a === "closeshift") closeShiftSheet(id);
    else if (a === "member") memberSheet(id);
    else if (a === "labels") labelsSheet();
  } catch (er) { toast(errMsg(er), true); b.disabled = false; }
});
document.addEventListener("input", e => {
  if (e.target.id === "q") { S.q = e.target.value; $("#plist").innerHTML = productRows(); }
});
document.addEventListener("submit", async e => {
  if (e.target.id !== "lf") return; e.preventDefault();
  const f = new FormData(e.target), btn = $("#lb"); btn.disabled = true;
  try { await api.signIn(f.get("u"), f.get("p")); await boot(); }
  catch (er) { toast(errMsg(er), true); btn.disabled = false; }
});

async function boot() {
  const u = await api.session();
  if (!u) { S.me = null; render(); return; }
  S.user = u;
  try { S.me = await api.me(u.id); } catch (_) { S.me = null; }
  if (!S.me) { await api.signOut(); toast("الحساب مش مظبوط. كلّم المالك.", true); render(); return; }
  if (!S.me.active) { await api.signOut(); S.me = null; render(); toast("حسابك لسه متفعّلش. كلّم المالك.", true); return; }
  S.loaded = false; render();
  try { await loadAll(); } catch (er) { toast(errMsg(er), true); S.loaded = true; }
  render();
}
boot();
