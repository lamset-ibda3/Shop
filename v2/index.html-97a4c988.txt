const nz = v => Number(v || 0);
const itemCost = i => { const c = Array.isArray(i.sale_item_costs) ? i.sale_item_costs[0] : i.sale_item_costs; return c ? nz(c.unit_cost) : null; };
function salesStats(list) {
  const st = { net: 0, count: 0, discounts: 0, cash: 0, vodafone: 0, instapay: 0, profit: 0, hasCost: true, returns: 0 };
  list.filter(s => !s.voided_at).forEach(s => {
    st.count++; st.net += nz(s.total); st.discounts += nz(s.discount) - nz(s.discount_returned); payList(s).forEach(p => st[p.method] += p.amount); st.profit += nz(s.discount_returned) - nz(s.discount);
    (s.sale_items || []).forEach(i => { const c = itemCost(i); if (c == null) st.hasCost = false; else st.profit += i.qty * (nz(i.unit_price) - c); if (i.qty < 0) st.returns += -i.qty * nz(i.unit_price); });
  });
  return st;
}

function cartCalc() {

  for (const l of c.lines) { const p = prodById(l.pid); if (!p) continue; sub += l.qty * nz(p.price); l.qty > 0 ? hasS = true : hasR = true; }

/* كاميرا الباركود: بتشتغل على كروم أندرويد */
async function startScan(onCode, once) {
  if (!("BarcodeDetector" in window) || !navigator.mediaDevices?.getUserMedia) return toast("الكاميرا مش مدعومة في المتصفح ده. اكتب الكود بإيدك.", true);
  let stream;
  try { stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: { ideal: "environment" } }, audio: false }); }
  catch (_) { return toast("مفيش إذن للكاميرا. فعّله من إعدادات المتصفح.", true); }
  const ov = document.createElement("div"); ov.className = "scan";
  ov.innerHTML = `<video playsinline muted></video><div class="frame"></div><div class="hint">وجّه الكاميرا على الباركود</div><button class="btn" id="scx">إغلاق</button>`;
  document.body.appendChild(ov);
  const v = $("video", ov); v.srcObject = stream; await v.play().catch(() => { });
  let det; try { det = new BarcodeDetector({ formats: ["code_128", "ean_13", "ean_8", "upc_a", "upc_e"] }); } catch (_) { det = new BarcodeDetector(); }
  let last = "", lt = 0, alive = true, iv;
  const stop = () => { alive = false; clearInterval(iv); stream.getTracks().forEach(t => t.stop()); ov.remove(); };
  $("#scx", ov).onclick = stop;
  iv = setInterval(async () => {
    if (!alive || v.readyState < 2) return;
    try {
      const r = await det.detect(v); if (!r.length) return;
      const code = r[0].rawValue, now = Date.now(); if (code === last && now - lt < 1800) return;
      last = code; lt = now; navigator.vibrate && navigator.vibrate(60);
      if (once) { stop(); onCode(code); } else { const msg = onCode(code); let f = $(".scanflash", ov); if (!f) { f = document.createElement("div"); f.className = "scanflash"; ov.appendChild(f); } f.textContent = msg || ""; }
    } catch (_) { }
  }, 220);
}

function payCalc(wid, month) {
  const es = S.pay.entries.filter(e => e.worker_id === wid && String(e.entry_date).slice(0, 7) === month);
  const sum = k => es.filter(e => e.kind === k).reduce((a, e) => a + nz(e.amount), 0);
  const salary = nz(S.pay.salaries[wid]), adv = sum("advance"), bonus = sum("bonus"), ded = sum("deduction"), paid = sum("salary_paid");
  return { es, salary, adv, bonus, ded, paid, left: salary + bonus - adv - ded - paid };
}

async function boot() {
  const u = await api.session();
  if (!u) { S.me = null; render(); return; }
  S.user = u;
  try { S.me = await api.me(u.id); } catch (_) { S.me = null; }
  if (!S.me) { await api.signOut(); toast("الحساب مش مظبوط. كلّم المالك.", true); render(); return; }
  if (!S.me.active) { await api.signOut(); S.me = null; render(); toast("حسابك لسه متفعّلش. كلّم المالك.", true); return; }
  S.loaded = false; render();
  try { await loadAll(); } catch (er) { toast(errMsg(er), true); S.loaded = true; }
  render();
}
